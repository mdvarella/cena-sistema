  var iaId=opts.ia_item_id||null;
  (_rhCtKit||[]).forEach(function(it){
    if(it===dest) return;
    var sameUrl=url && it.arquivo_url && String(it.arquivo_url)===String(url);
    var sameIa=iaId && it.ia_item_id && String(it.ia_item_id)===String(iaId);
    var sameNome=nome && it.arquivo_nome && String(it.arquivo_nome)===String(nome) && !it.documento_id;
    if(!(sameUrl||sameIa||sameNome)) return;
    it.arquivo_url=null;
    it.arquivo_nome=null;
    it.ia_item_id=null;
    if(['Recebido','Validado','Em conferencia','A vencer'].indexOf(it.status_kit)>=0){
      it.status_kit='Pendente';
      it.acao='Providenciar';
    }
  });
  dest.arquivo_url=url;
  dest.arquivo_nome=nome;
  dest.ia_item_id=iaId||dest.ia_item_id||null;
  if(['Nao se aplica','Dispensado','Pendente','','Atendido por documento equivalente'].indexOf(String(dest.status_kit||''))>=0
      || !dest.status_kit){
    dest.status_kit='Recebido';
  } else if(dest.status_kit!=='Validado'){
    dest.status_kit='Recebido';
  }
  dest.obrigatorio=true;
  dest.acao=opts.acao||'IA · no processo até efetivação';
  dest.motivo_exigencia=opts.motivo||('Classificado como '+tipo);
  if(_rhCtAtual && rhCtTipoEq(tipo,'CNH')){
    _rhCtAtual.possui_cnh=true;
    dest.motivo_exigencia='CNH anexada/classificada no processo';
  }
  return dest;
}
