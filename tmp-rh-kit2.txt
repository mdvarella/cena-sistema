async function rhKitColabAtual(){

  if(!colId) return null;
  var col=(colaboradores||[]).find(function(c){return String(c.id)===String(colId);})||null;

    var rows=await sbFetch('colaboradores',{filters:['id=eq.'+colId],limit:1});
    col=rows&&rows[0];
    if(col && colaboradores) colaboradores.push(col);

  return col;
}

function rhIaAbrirCapturaCamera(contexto, opts){
  opts = opts || {};
  if(!rhModuloPermitido('rh-entrada-ia') && !opts.bypassPerm){
    // permite chamada futura de outras telas RH se já autenticado no módulo
    if(!usuarioLogado){ progShowToast('Faça login.','erro'); return; }
  }
  _rhIaCam.contexto = contexto || 'dossie';
  _rhIaCam.opts = opts;
  _rhIaCam.mode = opts.mode || 'unico';
  _rhIaCam.pages = [];
  _rhIaCam.deviceIdx = 0;
  rhIaPararCamera();

  var hasGUM = !!(navigator.mediaDevices && navigator.mediaDevices.getUserMedia);
  if(!hasGUM){
    // fallback capture
    var inp = document.createElement('input');
    inp.type='file'; inp.accept='image/*'; inp.capture='environment'; inp.style.display='none';
    document.body.appendChild(inp);
    inp.onchange = function(){
      var f = inp.files && inp.files[0];
      document.body.removeChild(inp);
      if(f) rhIaReceberCapturaArquivo(f);
    };
    inp.click();
    return;
  }
  rhIaRenderModalCamera();
  rhIaListarCameras().then(function(){ return rhIaIniciarStream(); }).catch(function(e){
    progShowToast('Câmera indisponível: '+((e&&e.message)||e)+'. Usando seletor de arquivo.','erro');
    closeModal();
    var inp2 = document.createElement('input');
    inp2.type='file'; inp2.accept='image/*'; inp2.capture='environment';
    inp2.onchange=function(){ var f=inp2.files&&inp2.files[0]; if(f) rhIaReceberCapturaArquivo(f); };
    inp2.click();
  });
}
async function rhIaListarCameras(){
  try{
    var tmp = await navigator.mediaDevices.getUserMedia({audio:false, video:{width:{ideal:((typeof rhIaCapCfg==='function'&&rhIaCapCfg().w)||2560)}, height:{ideal:((typeof rhIaCapCfg==='function'&&rhIaCapCfg().h)||1920)}}});
    tmp.getTracks().forEach(function(t){t.stop();});
  }catch(e){}
  var all = await navigator.mediaDevices.enumerateDevices();
  _rhIaCam.devices = (all||[]).filter(function(d){ return d.kind==='videoinput'; });
  rhIaCapEscolherDevice();
}
async function rhIaIniciarStream(){
  rhIaPararCamera({keepPreview:true});
  var cfg = rhIaCapCfg();
  var video = {width:{ideal:cfg.w}, height:{ideal:cfg.h}};
  if(_rhIaCam.devices.length && _rhIaCam.devices[_rhIaCam.deviceIdx]){
    video.deviceId = {exact: _rhIaCam.devices[_rhIaCam.deviceIdx].deviceId};
  } else {
    video.facingMode = {ideal:_rhIaCam.facing};
  }
  try{
    _rhIaCam.stream = await navigator.mediaDevices.getUserMedia({audio:false, video:video});
  }catch(e1){
    var simple = video.deviceId ? {deviceId:video.deviceId} : {facingMode:{ideal:_rhIaCam.facing}};
    _rhIaCam.stream = await navigator.mediaDevices.getUserMedia({audio:false, video:simple});
  }
  var v = document.getElementById('rh-ia-cam-video');
  if(v){ v.srcObject = _rhIaCam.stream; v.play().catch(function(){}); }
  var lab = document.getElementById('rh-ia-cam-device-lab');
  if(lab){
    var d = _rhIaCam.devices[_rhIaCam.deviceIdx];
    lab.textContent = d && d.label ? d.label : ('Câmera '+(_rhIaCam.deviceIdx+1));
  }
  var all = rhIaCapLoad();
  var dev = rhIaCapDeviceId();
  if(dev && all.lastDeviceId!==dev){ all.lastDeviceId = dev; rhIaCapSave(all); }
  rhIaCapSincronizarSelects();
  rhIaCapAtualizarInfo();
}

  var cargo=rhCtVal('cargo'), funcao=rhCtVal('funcao')||cargo;

    if(rhCtVal(m.p)!==m.v) return;
