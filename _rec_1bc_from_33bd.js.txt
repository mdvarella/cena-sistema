function cenaTagNormalizarUid(valor){
  if(typeof cenaRfidNormalizarUid==='function') return cenaRfidNormalizarUid(valor);
  return String(valor==null?'':valor)
    .replace(/[\n\r\t]+/g,'')
    .replace(/[\s\-]+/g,'')
    .trim()
    .toUpperCase();
}

function cenaTagPodeCadastrar(){
  if(typeof perfilAcessoTotal==='function' && perfilAcessoTotal()) return true;
  var p = String((usuarioLogado && usuarioLogado.perfil) || '').toLowerCase();
  return p==='admin' || p==='diretoria' || p==='gestor' || p==='coordenador';
}

function cenaTagFmtTs(iso){
  if(typeof cenaRfidFmtTs==='function') return cenaRfidFmtTs(iso);
  if(!iso) return 'ÔÇö';
  var d = new Date(iso);
  if(isNaN(d.getTime())) return 'ÔÇö';
  function p(n){ return String(n).padStart(2,'0'); }
  return p(d.getDate())+'/'+p(d.getMonth()+1)+'/'+d.getFullYear()+' '+p(d.getHours())+':'+p(d.getMinutes());
}

function cenaTagEsc(s){
  if(typeof cenaRfidEsc==='function') return cenaRfidEsc(s);
  if(typeof escHtml==='function') return escHtml(String(s==null?'':s));
  return String(s==null?'':s).replace(/[&<>"']/g, function(ch){
    return ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'})[ch];
  });
}

function cenaTagVeiculoMotivo(vei){
  if(!vei || !vei.id) return 'VEICULO_INATIVO';
  if(vei.ativo===false) return 'VEICULO_INATIVO';
  var st=String(vei.status||'').toLowerCase();
  if(st==='baixado' || st==='inativo' || st.indexOf('inativ')>=0) return 'VEICULO_INATIVO';
  if(st==='bloqueado' || st==='quarentena') return 'VEICULO_BLOQUEADO';
  return null;
}

function cenaTagParseErro(txt, status){
  var msg = 'Erro TAG ('+status+')';
  try{
    var j = JSON.parse(txt);
    msg = j.message || j.details || j.hint || msg;
    if(String(j.hint||'')==='TAG_UID_HISTORICO' || /j├í vinculad/i.test(msg)){
      if(msg.indexOf('Ôøö')!==0) msg = 'Ôøö '+msg;
    }
  }catch(e){
    if(txt) msg = String(txt).slice(0, 280);
  }
  return msg;
}

async function cenaTagRpc(fn, params){
  if(DEMO) return {ok:false, error:'DEMO_USE_LOCAL'};
  var tok = null;
  try{
    tok = await cenaAuthGetValidToken({requireAuth:true, forceLogin:false, reason:'tag:'+fn});
  }catch(e){ tok = null; }
  if(!tok || tok==='demo' || (typeof sbAuthJwtIsAuthenticated==='function' && !sbAuthJwtIsAuthenticated(tok))){
    return {ok:false, error:'Sess├úo Auth obrigat├│ria para TAG. Fa├ºa login autenticado (n├úo MD5).'};
  }
  try{
    var r = await fetch(SB.url+'/rest/v1/rpc/'+fn, {
      method:'POST',
      headers:{
        'apikey': SB.key,
        'Authorization': 'Bearer '+tok,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(params||{})
    });
    var txt = await r.text();
    var data = null;
    try{ data = txt ? JSON.parse(txt) : null; }catch(e1){ data = null; }
    if(!r.ok){
      console.error('cenaTagRpc', fn, r.status, txt);
      return {ok:false, error: cenaTagParseErro(txt, r.status), status:r.status, raw:data};
    }
    return {ok:true, data:data};
  }catch(e){
    console.error('cenaTagRpc', fn, e);
    return {ok:false, error: String(e && e.message || e)};
  }
}

function cenaTagDemoResolver(uid){
  var n = cenaTagNormalizarUid(uid);
  if(!n) return {ok:false, encontrado:false, autenticavel:false, motivo:'UID_VAZIO', credencial:null, veiculo:null};
  var cred = _cenaTagDemoCreds.filter(function(x){ return x.tipo==='RFID_TAG' && x.uid_normalizado===n; })[0];
  if(!cred) return {ok:true, encontrado:false, autenticavel:false, motivo:'TAG_DESCONHECIDA', credencial:null, veiculo:null};
  var vei = (typeof frt_veiculos!=='undefined'?frt_veiculos:[]).find(function(v){ return String(v.id)===String(cred.veiculo_id); }) || {};
  var motivoCred = cred.status==='BLOQUEADO' ? 'TAG_BLOQUEADA' : (cred.status==='SUBSTITUIDO' ? 'TAG_SUBSTITUIDA' : null);
  var motivoVei = cred.status==='ATIVO' ? cenaTagVeiculoMotivo(vei) : null;
  var motivo = motivoCred || motivoVei;
  var aut = !motivo;
  return {
    ok:true, encontrado:true, autenticavel:!!aut,
    motivo: motivo,
    credencial:{id:cred.id, status:cred.status},
    veiculo:{id:vei.id||cred.veiculo_id, placa:vei.placa||'ÔÇö', modelo:vei.modelo||'ÔÇö', status:vei.status||'ÔÇö'}
  };
}

function cenaTagDemoSubstituir(veicId, uid, motivo){
  var n = cenaTagNormalizarUid(uid);
  if(!n) return {ok:false, error:'UID TAG vazio ap├│s normaliza├º├úo'};
  var at = cenaTagDemoAtiva(veicId);
  if(!at) return {ok:false, error:'Ve├¡culo n├úo possui TAG ATIVA para substituir'};
  if(at.uid_normalizado===n) return {ok:false, error:'O UID lido j├í ├® a TAG ATIVA deste ve├¡culo'};
  var exist = _cenaTagDemoCreds.filter(function(x){ return x.tipo==='RFID_TAG' && x.uid_normalizado===n; })[0];
  if(exist && String(exist.veiculo_id)!==String(veicId)) return {ok:false, error:'Ôøö TAG j├í vinculada a outro ve├¡culo.'};
  at.status='SUBSTITUIDO';
  var now=new Date().toISOString();
  var nova;
  if(exist && String(exist.veiculo_id)===String(veicId)){
    exist.status='ATIVO';
    exist.ativado_em=now;
    exist.substitui_credencial_id=at.id;
    exist.bloqueado_em=null;
    exist.motivo_bloqueio=null;
    nova=exist;
  } else {
    nova={
      id:'tag-demo-'+Date.now(), veiculo_id:veicId, tipo:'RFID_TAG',
      uid:String(uid||'').trim(), uid_normalizado:n, status:'ATIVO',
      ativado_em:now, criado_em:now, substitui_credencial_id:at.id,
      origem_leitura:'SIMULADOR', criado_por:'DEMO', motivo:motivo||null
    };
    _cenaTagDemoCreds.push(nova);
  }
  return {ok:true, data:{ok:true, acao:'TAG_SUBSTITUIDA', credencial_anterior:{id:at.id, status:'SUBSTITUIDO'}, credencial:{id:nova.id, status:'ATIVO'}}};
}

async function cenaAcessoInfraSetStatus(id, status){
  if(!cenaAcessoInfraPodeCadastrar()){ cenaAcessoInfraToast('Sem permiss├úo para alterar status.','erro'); return; }
  var ok = ['SIMULADO','ONLINE','OFFLINE','ERRO'].indexOf(status)>=0;
  if(!ok){ cenaAcessoInfraToast('Status inv├ílido.','erro'); return; }
  var patch = {status_comunicacao:status};
  if(status==='ONLINE') patch.ultima_comunicacao = new Date().toISOString();
  if(typeof DEMO!=='undefined' && DEMO){
    var d = _cenaAcessoDisps.find(function(x){ return String(x.id)===String(id); });
    if(d){ d.status_comunicacao=status; if(patch.ultima_comunicacao) d.ultima_comunicacao=patch.ultima_comunicacao; }
    cenaAcessoInfraRender();
    cenaAcessoInfraToast('Status '+status+' (simulado ÔÇö sem hardware).');
    return;
  }
  try{
    await sbUpdate('portaria_dispositivos', patch, 'id=eq.'+id);
    var row = _cenaAcessoDisps.find(function(x){ return String(x.id)===String(id); });
    if(row){
      row.status_comunicacao=status;
      if(patch.ultima_comunicacao) row.ultima_comunicacao=patch.ultima_comunicacao;
    }
    cenaAcessoInfraRender();
    cenaAcessoInfraToast('Status '+status+' (administrativo ÔÇö sem hardware).');
  }catch(e){
    cenaAcessoInfraToast((e&&e.message)||'Falha ao alterar status.','erro');
  }
}

function fmtD(iso){
  if(iso==null||iso==='') return 'ÔÇö';
  try{
    var s=String(iso).trim();
    if(!s) return 'ÔÇö';
    if(s.indexOf('T')>=0) s=s.split('T')[0];
    else if(/^\d{2}\/\d{2}\/\d{4}/.test(s)){
      var p=s.split('/');
      s=p[2]+'-'+p[1]+'-'+p[0];
    } else {
      s=s.slice(0,10);
    }
