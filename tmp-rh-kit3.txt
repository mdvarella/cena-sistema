function rhCtAnexarKit(inp){
  var tipo=decodeURIComponent((inp&&inp.getAttribute('data-tipo'))||'');
  var file=inp&&inp.files&&inp.files[0];
  if(inp) inp.value='';
  if(!file) return;
  return rhCtAnexarKitArquivo(tipo, file);
}

function rhIaUsarFoto(){
  if(!_rhIaCam.previewBlob) return;
  var papel = 'unico', pagina = 1;
  if(_rhIaCam.mode==='frente_verso'){
    papel = _rhIaCam.pages.length===0 ? 'frente' : 'verso';
    pagina = _rhIaCam.pages.length+1;
  } else if(_rhIaCam.mode==='multipagina'){
    papel = 'pagina';
    pagina = _rhIaCam.pages.length+1;
  }
  _rhIaCam.pages.push({blob:_rhIaCam.previewBlob, papel:papel, pagina:pagina});
  _rhIaCam.waitingConfirm=false;
  if(_rhIaCam.mode==='unico'){
    rhIaFinalizarGrupoCamera();
    return;
  }
  if(_rhIaCam.mode==='frente_verso' && _rhIaCam.pages.length>=2){
    var fin=document.getElementById('rh-ia-cam-finish');
    if(fin) fin.style.display='';
    rhIaRefazerFoto();
    progShowToast('Frente e verso capturados. Clique em Finalizar documento.');
    return;
  }
  rhIaRefazerFoto();
  var fin2=document.getElementById('rh-ia-cam-finish');
  if(fin2 && _rhIaCam.mode==='multipagina' && _rhIaCam.pages.length>=1) fin2.style.display='';
  progShowToast('Página '+_rhIaCam.pages.length+' adicionada. Capture outra ou finalize.');
}

function rhIaFinalizarGrupoCamera(){
  if(!_rhIaCam.pages.length && _rhIaCam.previewBlob){
    _rhIaCam.pages.push({blob:_rhIaCam.previewBlob, papel:'unico', pagina:1});
  }
  if(!_rhIaCam.pages.length){ progShowToast('Nenhuma foto capturada.','erro'); return; }
  var grupoId = (typeof gerarUUID==='function'?gerarUUID():('grp'+Date.now()));
  var files = _rhIaCam.pages.map(function(p,i){
    var nome = 'camera_'+(_rhIaCam.contexto||'dossie')+'_'+(p.papel||'pag')+'_'+p.pagina+'_'+Date.now()+'.jpg';
    var f;
    try{ f = new File([p.blob], nome, {type:'image/jpeg'}); }
    catch(eFile){
      f = p.blob;
      try{ f.name = nome; }catch(e2){}
    }
    f._rhIaMeta = {
      origem_entrada:'camera',
      contexto_origem: (_rhIaCam.opts&&_rhIaCam.opts.contexto_origem) || (_rhIaCam.contexto==='contratacao'||_rhIaCam.contexto==='promocao'?_rhIaCam.contexto:((_rhIaContextoPadrao&&_rhIaContextoPadrao.contexto_origem)||'dossie')),
      colaborador_id_contexto: (_rhIaCam.opts&&_rhIaCam.opts.colaborador_id) || (_rhIaContextoPadrao&&_rhIaContextoPadrao.colaborador_id_contexto) || null,
      contratacao_id: (_rhIaCam.opts&&_rhIaCam.opts.contratacao_id) || (_rhIaContextoPadrao&&_rhIaContextoPadrao.contratacao_id) || null,
      candidatura_id: (_rhIaCam.opts&&_rhIaCam.opts.candidatura_id) || (_rhIaContextoPadrao&&_rhIaContextoPadrao.candidatura_id) || null,
      vaga_id: (_rhIaCam.opts&&_rhIaCam.opts.vaga_id) || (_rhIaContextoPadrao&&_rhIaContextoPadrao.vaga_id) || null,
      cargo_id: (_rhIaCam.opts&&_rhIaCam.opts.cargo_id) || (_rhIaContextoPadrao&&_rhIaContextoPadrao.cargo_id) || null,

      grupo_papel: p.papel||'unico',

  });

  closeModal();

    try{ _rhIaCam.opts.onFiles(files); }catch(eOn){ progShowToast((eOn&&eOn.message)||'Falha ao usar captura.','erro'); }
    return;
  }
  rhIaAdicionarArquivos(files);
  progShowToast(files.length+' captura(s) na seleção. Envie para staging.');
}

async function rhIaMapComConcorrencia(items, limit, worker){
  var i = 0, ativos = 0, resultados = new Array(items.length);
  return new Promise(function(resolve){
    function next(){
      while(ativos < limit && i < items.length){
        (function(idx){
          ativos++;
          Promise.resolve(worker(items[idx], idx)).then(function(res){
            resultados[idx]=res;
          }).catch(function(err){
            resultados[idx]={ok:false, erro:(err&&err.message)||String(err)};
          }).then(function(){
            ativos--;
            if(i >= items.length && ativos===0) resolve(resultados);
            else next();
          });
        })(i++);
      }
      if(items.length===0) resolve([]);
    }
    next();
  });
}

function rhIaRemoverSelecao(idx){
  if(_rhIaUploadBusy){ progShowToast('Aguarde o upload terminar.','erro'); return; }
  _rhIaSelecao.splice(idx,1);
  rhRenderEntradaIa();
}
function rhIaLimparSelecao(){
  if(_rhIaUploadBusy){ progShowToast('Aguarde o upload terminar.','erro'); return; }
  if(_rhIaProcBusy){ progShowToast('Aguarde o processamento da IA.','erro'); return; }
  var n = (_rhIaSelecao||[]).length;
  if(!n){ progShowToast('Seleção já está vazia.'); return; }
  _rhIaSelecao = [];
  progShowToast(n===1 ? '1 arquivo removido da seleção.' : (n+' arquivos removidos da seleção.'));
  rhRenderEntradaIa();
}
